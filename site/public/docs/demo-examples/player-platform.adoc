== PlayerPlatform Integration of HPlayer

Link to the playerPlatform have a lifespan of 24h , player on the left will not work.
But the code example is functional.

*Don't forget to add mux.js for hls support.*

[source,html]
----
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>HPlayer</title>
    <!-- HPlayer compiled CSS library: -->
    <link rel="stylesheet" href="css/index.css">

</head>
<body>
<!-- Container for the UI overlay-->
<div class="player-container">
    <!-- HTMLVideoElement, width and height to 100% required to fill the player-container -->
    <video class="h-player" style="width:100%;height:100%"></video>
</div>
<!-- HPlayer compiled JS library: -->
    <script src="js/HPlayer.bundle.js"></script>
    <!-- For hls support -->
    <script defer src="https://github.com/videojs/mux.js/releases/download/v5.13.0/mux.min.js"></script>
<script>
const videoContainer = document.querySelector('.player-container')
const video = document.querySelector('.h-player');
const options = {
  basic: {
    source: 'player-platform:20210901_33951609_00000HR', // Player platform video id
  },
  hexa_configurations: {
    player_platform: { // Player platform backend description/configuration
      endpoint: 'https://equidia-vodce-p-api.hexaglobe.net/video/info',
      publicKey: 'b16d48525feb78bb5f729ab781e2b3cc',
      hash: '03bbf347999d0e36882bcb3be9761e57ecf84af7f7922a81b76826edc014aa8e',
      expiresAt: '1631785491',
    },
  },
  ui_configuration: {
    addBigPlayButton: true,
    controlPanelElements: [
      'play_pause',
      'time_and_duration',
      'spacer',
      'mute',
      'volume',
      'fullscreen',
      'overflow_menu' // Enable the overflow menu to display the resolution select(only for hls)
    ],
    overflowMenuButtons: [
      'captions',
      'quality', // Resolution select (only for hls)
      'language',
      'picture_in_picture',
      'cast',
      'playback_rate',
    ],
  },
}


const hplayer = new HPlayer(video,videoContainer, options);
hplayer.init_();
</script>
</body>

</html>

----

=== Changing video dynamically

To change video displayed without reloading the whole player.

[source,javascript]
----
// This will use the hexa_configurations.player_platform configuration given when doing new HPlayer()
hplayer.loadAssets_({
  basic: {
      source : 'player-platform:MY_VIDEO_ID'
  }
});
----
