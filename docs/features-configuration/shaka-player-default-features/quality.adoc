= Quality Feature

- Adds a button that controls enabling/disabling of abr and video resolution selection.

== How to configure and enable this feature

[source, json]
----
{
  "ui_configuration": {
    "controlPanelElements": [
      "overflow_menu"
    ],
    "overflowMenuButtons": [
      "quality"
    ]
  }
}
----

=== Example

[source,html]
----
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Hexa Player</title>
<!-- Hexaglobe Player compiled CSS library: -->
    <link rel="stylesheet" href="css/index.css">

</head>
<body>
<!-- Container for the UI overlay-->
<div class="hexa-container">
    <!-- HTMLVideoElement, width and height to 100% required to fill the hexa-container -->
    <video class="hexa-player" style="width:100%;height:100%"></video>
</div>
<!-- Hexaglobe Player compiled JS library: -->
<script src="js/HexaglobePlayer.bundle.js"></script>
<script>
const videoContainer = document.querySelector('.hexa-container')
const video = document.querySelector('.hexa-player');
const options = {
  basic: {
    source: "http://commondatastorage.googleapis.com/gtv-videos-bucket/sample/BigBuckBunny.mp4",
  },
  ui_configuration: {
    controlPanelElements: [
      "overflow_menu"
    ],
    overflowMenuButtons: [
      "quality"
    ]
  }
};

const hexaPlayer = new HexaglobePlayer(video,videoContainer, options);
hexaPlayer.init_();
</script>
</body>

</html>

----
